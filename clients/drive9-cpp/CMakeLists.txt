function(drive9_header name url sha)
  set(path "${CMAKE_CURRENT_BINARY_DIR}/deps/${name}")
  if(EXISTS "${path}")
    file(SHA256 "${path}" actual)
  endif()
  if(NOT actual STREQUAL sha)
    file(DOWNLOAD "${url}" "${path}.tmp" TLS_VERIFY ON TIMEOUT 60
      EXPECTED_HASH "SHA256=${sha}" STATUS result)
    list(GET result 0 code)
    if(NOT code EQUAL 0)
      message(FATAL_ERROR "Unable to fetch pinned dependency ${name}: ${result}")
    endif()
    file(RENAME "${path}.tmp" "${path}")
  endif()
endfunction()
